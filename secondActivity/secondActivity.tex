\begin{theorem}
    If $f'(x) >0$ on $I$ then $f$ is increasing on $I$, and\\
    if $f'(x) <0$ on $I$ then $f$ is decreasing on $I$.
\end{theorem}

\begin{theorem}
    If $f''(x) >0$ on $I$ then $f$ is concave up on $I$, and\\
    if $f''(x) <0$ on $I$ then $f$ is concave down on $I$.
\end{theorem}
